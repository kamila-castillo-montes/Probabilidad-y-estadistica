# CASO: Telemetría de Lead Time for Changes (Horas)
# ==============================================================================

# 1. DEPENDENCIAS
# 'moments' se emplea para el cálculo formal de asimetría y curtosis.
if (!requireNamespace("moments", quietly = TRUE)) {
  install.packages("moments")
}
library(moments)

# Fijamos semilla para asegurar reproducibilidad exacta
set.seed(123)

# ==============================================================================
# 2. GENERACIÓN DE DATOS SIMULADOS (N = 5,000)
# ==============================================================================

n_total <- 5000

# Representamos dos procesos operativos distintos (Distribución Bimodal):
# Subpoblación 1: Cambios estándar automatizados (Media ~20h, SD ~4)
# Subpoblación 2: Cambios complejos / bloqueados por QA manual (Media ~80h, SD ~10)
n_grupo1 <- 3000
n_grupo2 <- 1600
n_anomalos <- 400

# Generamos variables enteras mediante redondeo
lead_g1 <- round(rnorm(n_grupo1, mean = 20, sd = 4))
lead_g2 <- round(rnorm(n_grupo2, mean = 80, sd = 10))

# Introducimos datos incorrectos/inconsistencias:
# - Valores negativos (errores de sincronización de reloj NTP / cálculo de timestamp)
inconsistencias <- round(runif(n_anomalos / 2, min = -60, max = -1))

# Introducimos valores atípicos extremos (Outliers):
# - Despliegues bloqueados artificialmente durante meses (ej. 200h a 600h)
outliers_extremos <- round(runif(n_anomalos / 2, min = 200, max = 600))

# Unificamos el vector de métricas en valores enteros puros
lead_time_vector <- as.integer(c(lead_g1, lead_g2, inconsistencias, outliers_extremos))

# Generamos IDs de despliegue con duplicados simulando reintentos de red o fallas de webhooks
ids_con_duplicados <- sample(1:4200, size = n_total, replace = TRUE)

# Construcción del DataFrame sucio
df_devops <- data.frame(
  deployment_id = ids_con_duplicados,
  lead_time_hours = lead_time_vector
)

# Forzamos duplicados exactos en filas completas
filas_a_duplicar <- sample(1:nrow(df_devops), size = 60)
df_devops[filas_a_duplicar, ] <- df_devops[sample(filas_a_duplicar), ]

# Confirmamos estructura y tamaño exacto
dim(df_devops)
head(df_devops, 10)

# ==============================================================================
# 3. DIAGNÓSTICO DE INTEGRIDAD DE LOS DATOS
# ==============================================================================

# Conteo de registros duplicados exactos
duplicados_totales <- sum(duplicated(df_devops))
cat("Total de registros duplicados:", duplicados_totales, "\n")

# Conteo de inconsistencias lógicas (tiempos negativos)
negativos_totales <- sum(df_devops$lead_time_hours < 0)
cat("Total de inconsistencias lógicas (< 0h):", negativos_totales, "\n")

# ==============================================================================
# 4. MEDIDAS DE TENDENCIA CENTRAL Y FORMA DE LA DISTRIBUCIÓN
# ==============================================================================

x <- df_devops$lead_time_hours

# R base no incluye una función nativa paramétrica para la moda estadística
calcular_moda <- function(valores) {
  tabla_unicos <- table(valores)
  moda_val <- as.integer(names(tabla_unicos)[which.max(tabla_unicos)])
  return(moda_val)
}

# Cálculos estadísticos descriptivos
media_val    <- mean(x)
mediana_val  <- median(x)
moda_val     <- calcular_moda(x)
sd_val       <- sd(x)
iqr_val      <- IQR(x)
asimetria    <- moments::skewness(x)
curtosis_val <- moments::kurtosis(x)

# Consolidamos las métricas en un DataFrame para su lectura
tabla_metricas <- data.frame(
  Metrica = c("Media", "Mediana", "Moda", "Desviación Estándar", 
              "Rango Intercuartílico (IQR)", "Asimetría (Skewness)", "Curtosis"),
  Valor = c(media_val, mediana_val, moda_val, sd_val, iqr_val, asimetria, curtosis_val)
)

print(tabla_metricas)

# ==============================================================================
# 5. TABLA DE DISTRIBUCIÓN DE FRECUENCIAS
# ==============================================================================

# Definimos clases o intervalos de 25 horas para abarcar desde los negativos hasta los extremos
ancho_clase <- 25
cortes <- seq(min(x) - 1, max(x) + ancho_clase, by = ancho_clase)
intervalos <- cut(x, breaks = cortes, right = TRUE)

# Frecuencia Absoluta (f_i): Número de observaciones por intervalo
frec_abs <- table(intervalos)

# Frecuencia Relativa (h_i): Proporción respecto al total
frec_rel <- prop.table(frec_abs)

# Frecuencias Acumuladas (F_i y H_i)
frec_abs_acum <- cumsum(frec_abs)
frec_rel_acum <- cumsum(frec_rel)

# Construcción de la tabla completa de distribución de frecuencias
tabla_frecuencias <- data.frame(
  Intervalo_Horas    = names(frec_abs),
  Frec_Absoluta      = as.numeric(frec_abs),
  Frec_Relativa      = round(as.numeric(frec_rel), 4),
  Frec_Abs_Acumulada = as.numeric(frec_abs_acum),
  Frec_Rel_Acumulada = round(as.numeric(frec_rel_acum), 4)
)

# Imprimimos los intervalos con observaciones
tabla_frecuencias_filtrada <- tabla_frecuencias[tabla_frecuencias$Frec_Absoluta > 0, ]
print(tabla_frecuencias_filtrada, row.names = FALSE)
